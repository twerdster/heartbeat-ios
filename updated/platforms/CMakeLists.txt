cmake_minimum_required(VERSION 3.20)
project(heartbeat_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(HEARTBEAT_BUNDLE_ID "io.github.twerdster.heartbeat.updated" CACHE STRING "Use an identifier registered to your Apple team")
option(HEARTBEAT_NATIVE_SANITIZE "Check portable filter/state machine with ASan/UBSan" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_library(heartbeat_core SHARED heartbeat_c.cpp ../native/Butterworth.cpp)
 target_include_directories(heartbeat_core PUBLIC . ../../source/Heartbeat)
 target_compile_options(heartbeat_core PRIVATE -ffp-contract=off)
 add_executable(heartbeat_cli heartbeat_cli.cpp ../native/Butterworth.cpp)
 target_include_directories(heartbeat_cli PRIVATE . ../../source/Heartbeat)
 target_compile_options(heartbeat_cli PRIVATE -ffp-contract=off)
 if(HEARTBEAT_NATIVE_SANITIZE)
  foreach(target heartbeat_core heartbeat_cli)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
 endif()
endif()
if(APPLE)
 enable_language(OBJCXX)
 set(DATA data/bidmc01-ppg.csv data/NOTICE.txt)
 add_executable(HeartbeatUpdated MACOSX_BUNDLE app.mm ../native/Butterworth.cpp ${DATA})
 set_source_files_properties(${DATA} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
 target_include_directories(HeartbeatUpdated PRIVATE . ../../source/Heartbeat)
 target_compile_options(HeartbeatUpdated PRIVATE -fobjc-arc -ffp-contract=off)
 target_link_libraries(HeartbeatUpdated PRIVATE "-framework Foundation" "-framework CoreGraphics" "-framework QuartzCore")
 if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_link_libraries(HeartbeatUpdated PRIVATE "-framework UIKit")
  set_target_properties(HeartbeatUpdated PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist" XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${HEARTBEAT_BUNDLE_ID}" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1" XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Automatic")
 else()
  set_target_properties(HeartbeatUpdated PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "HeartBeat Updated" MACOSX_BUNDLE_GUI_IDENTIFIER "${HEARTBEAT_BUNDLE_ID}")
  target_link_libraries(HeartbeatUpdated PRIVATE "-framework Cocoa")
 endif()
endif()
