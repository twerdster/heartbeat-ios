cmake_minimum_required(VERSION 3.20)
project(heartbeat_repaired_filter LANGUAGES CXX)
option(HEARTBEAT_SANITIZE "Check the bounded filter with address/undefined sanitizers" OFF)
add_library(heartbeat_filter STATIC Butterworth.cpp)
target_include_directories(heartbeat_filter PUBLIC ../../source/Heartbeat)
target_compile_features(heartbeat_filter PUBLIC cxx_std_17)
add_executable(filter_regression filter_regression.cpp)
target_link_libraries(filter_regression PRIVATE heartbeat_filter)
if(HEARTBEAT_SANITIZE)
  foreach(target heartbeat_filter filter_regression)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME bounded_filter COMMAND filter_regression)
